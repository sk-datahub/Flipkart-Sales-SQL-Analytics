01BASICS

Calculate total revenue

Find the total sales revenue generated across all months.

SELECT
    SUM(monthly_sales) AS total_revenue
FROM monthly_sales;

02TREND

Monthly revenue report

Compare revenue month by month in chronological order.

SELECT
    sales_month,
    monthly_sales AS revenue
FROM monthly_sales
ORDER BY sales_month;
SELECT · ORDER BY
03PROFIT

Top 5 profit months

Identify the five months with the highest profit.

SELECT
    sales_month,
    monthly_profit
FROM monthly_sales
ORDER BY monthly_profit DESC
LIMIT 5;
ORDER BY · LIMIT
04KPI

Calculate profit margin

Calculate the profit margin percentage for each month.

SELECT
    sales_month,
    ROUND(
        monthly_profit / NULLIF(monthly_sales, 0) * 100,
        2
    ) AS profit_margin_pct
FROM monthly_sales
ORDER BY sales_month;
ROUND · NULLIF
05ORDERS

Order summary

View total orders and average orders per month.

SELECT
    SUM(total_orders) AS total_orders,
    ROUND(AVG(total_orders), 2) AS avg_orders_per_month
FROM monthly_sales;
SUM · AVG
06WINDOW

Rank months by revenue

Rank each month from highest to lowest revenue.

SELECT
    sales_month,
    monthly_sales,
    RANK() OVER (
        ORDER BY monthly_sales DESC
    ) AS revenue_rank
FROM monthly_sales;
RANK · OVER
07WINDOW

Month-over-month change

Compare revenue with the previous month.

WITH monthly_trend AS (
    SELECT
        sales_month,
        monthly_sales,
        LAG(monthly_sales) OVER (
            ORDER BY sales_month
        ) AS previous_month_sales
    FROM monthly_sales
)
SELECT *,
    ROUND(monthly_sales - previous_month_sales, 2)
        AS revenue_change
FROM monthly_trend
ORDER BY sales_month;
CTE · LAG
08SUBQUERY

Months above average sales

Find months where revenue is higher than the overall monthly average.

SELECT sales_month, monthly_sales
FROM monthly_sales
WHERE monthly_sales > (
    SELECT AVG(monthly_sales)
    FROM monthly_sales
)
ORDER BY monthly_sales DESC;
WHERE · Subquery
09LOGIC

Categorize sales performance

Label each month as high, medium, or low revenue using thresholds you can adjust.

SELECT
    sales_month,
    monthly_sales,
    CASE
        WHEN monthly_sales >= 7000000 THEN 'High'
        WHEN monthly_sales >= 5000000 THEN 'Medium'
        ELSE 'Low'
    END AS sales_category
FROM monthly_sales
ORDER BY sales_month;
CASE · WHEN
10SUMMARY

Overall business summary

Bring the main business KPIs together in one result row.

SELECT
    SUM(monthly_sales) AS total_revenue,
    SUM(monthly_profit) AS total_profit,
    SUM(total_orders) AS total_orders,
    ROUND(
        SUM(monthly_profit) / NULLIF(SUM(monthly_sales), 0) * 100,
        2
    ) AS profit_margin_pct
FROM monthly_sales;
